==================

  pa (int i = 0; i < 3; i += 1) {

  }
  pa (i = 0; i < 3) {
    kontinua;
  }
  si (int x = 1; x > 0) {
  } sinau si (x == 0) {
  }
}

  }
  si (int) 2.5 == 2 {
