fn metadi(int a, int b) num : Erru {
  si b == 0 {
    lansa Erru{mensagem = "zero"};
  }
  divolvi a / b;
}

fn valida(int n) : Erru {
  spia metadi(n, 2);
  num r = spia metadi(n, 0);
}
