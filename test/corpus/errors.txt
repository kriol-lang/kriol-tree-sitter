Fallible functions, tenta, lansa and sinon
==================

fn metadi(int a, int b) num : Erru {
  si b == 0 {
    lansa Erru::{mensage: "zero"};
  }
  divolvi a / b;
}

fn valida(int n) : Erru {
  tenta metadi(n, 2);
  num r = metadi(n, 0) sinon 1.5;
}
